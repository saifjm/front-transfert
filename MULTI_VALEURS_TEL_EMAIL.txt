┌──────────────────────────────────────────────────────────────────────────────┐
│ 📞 TÉLÉPHONE - Validation Multi-Numéros                                      │
└──────────────────────────────────────────────────────────────────────────────┘

  ✅ FORMATS ACCEPTÉS (par numéro)
  ─────────────────────────────────
  
  Format 1 : 8 chiffres
    Exemple : 12345678
  
  Format 2 : +216 suivi de 8 chiffres
    Exemple : +21612345678
  
  Format 3 : 00216 suivi de 8 chiffres
    Exemple : 0021612345678

  ✅ SAISIE MULTIPLE
  ─────────────────
  
  Séparez les numéros par un point-virgule (;)
  Les espaces avant/après sont automatiquement supprimés
  
  📝 EXEMPLES VALIDES
  ────────────────────
  
  ✓ 12345678
  ✓ +21612345678
  ✓ 12345678;98765432
  ✓ +21612345678;+21687654321
  ✓ 12345678; +21687654321; 0021611223344
  ✓ 12345678 ; 98765432 ; +21611223344
  
  ❌ EXEMPLES INVALIDES
  ──────────────────────
  
  ✗ 1234567              (7 chiffres seulement)
  ✗ +215123456789        (indicatif incorrect)
  ✗ 12345678;abc         (deuxième valeur non numérique)
  ✗ +216123456;98765432  (premier numéro incomplet)

  🎯 VALIDATION
  ─────────────
  
  • Chaque numéro est validé individuellement
  • Message d'erreur précise quel numéro est invalide
  • Exemple : "Numéro 2 invalide ("+215123456"). Format attendu : ..."
  • Compteur affiche : "✓ 3 numéro(s) valide(s)"

│ 💻 IMPLÉMENTATION TECHNIQUE                                                  │
└──────────────────────────────────────────────────────────────────────────────┘

  📁 Fichier : /components/AVAForm.tsx
  
  🔧 FONCTION : validateTelephoneTunisien(tel: string): string
  ──────────────────────────────────────────────────────────────
  
  const validateTelephoneTunisien = (tel: string): string => {
    if (!tel) return '';
    
    // Séparer et nettoyer
    const numeros = tel.split(';')
                       .map(num => num.trim())
                       .filter(num => num !== '');
    
    if (numeros.length === 0) return '';
    
    // Regex pour chaque format
    const regex8Digits = /^\d{8}$/;
    const regexPlus216 = /^\+216\d{8}$/;
    const regex00216 = /^00216\d{8}$/;
    
    // Valider chaque numéro
    for (let i = 0; i < numeros.length; i++) {
      const numero = numeros[i];
      if (!regex8Digits.test(numero) && 
          !regexPlus216.test(numero) && 
          !regex00216.test(numero)) {
        return `Numéro ${i + 1} invalide ("${numero}"). ` +
               `Format attendu : 8 chiffres, +216XXXXXXXX ou 00216XXXXXXXX`;
      }
    }
    
    return '';
  };
  
  🔧 FONCTION : validateEmail(email: string): string
  ──────────────────────────────────────────────────
  
  const validateEmail = (email: string): string => {
    if (!email) return '';
    
    // Séparer et nettoyer
    const emails = email.split(';')
                        .map(em => em.trim())
                        .filter(em => em !== '');
    
    if (emails.length === 0) return '';
    
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    
    // Valider chaque email
    for (let i = 0; i < emails.length; i++) {
      const em = emails[i];
      if (!emailRegex.test(em)) {
        return `Email ${i + 1} invalide ("${em}")`;
      }
    }
    
    return '';
  };

  Interface DTO : InitiationOuvertureDTO
  ──────────────────────────────────────
  
  interface InitiationOuvertureDTO {
    // ... autres champs
    tel?: string;      // Exemple : "12345678;+21687654321;0021611223344"
    email?: string;    // Exemple : "contact@test.com;admin@test.com"
  }
  
  📤 Envoi à l'API
  ────────────────
  Les valeurs sont envoyées telles quelles dans le DTO.
  C'est le backend qui sera responsable de parser les valeurs
  séparées par point-virgule si nécessaire.
  
  Exemple JSON :
  {
    "tel": "12345678;+21687654321",
    "email": "contact@exemple.com;admin@exemple.com",
    ...
  }

│ 📝 NOTES IMPORTANTES                                                         │
└──────────────────────────────────────────────────────────────────────────────┘

  ⚠️  Séparateur UNIQUEMENT le point-virgule (;)
      Les virgules (,) ou autres caractères ne sont pas supportés
  
  ⚠️  Validation stricte
      Tous les numéros/emails doivent être valides
      Un seul invalide bloque la soumission
  
  ⚠️  Type du champ email changé
      Le type HTML est passé de "email" à "text" pour permettre
      la saisie de plusieurs emails avec point-virgule
  
  ✅  Rétrocompatibilité
      Les anciennes saisies avec une seule valeur fonctionnent toujours
      Aucune migration de données nécessaire
