  CONTROLE_RNE (p_str in varchar2) return number IS
    vecteur varchar2(23);
    cleMF   varchar2(1);
    i       number;
    somme   number;
  BEGIN
    -- le vecteur des differentes cles
    vecteur := 'ABCDEFGHJKLMNPQRSTVWXYZ';
    
    if length(p_str) <> 8 then
      -- la chaine en entree est de long <> 8 alors non valide
      return 1;
    elsif p_str = '            ' then
      -- la chaine en entree est vide alors consideree comme valide
      return 0;
    elsif to_number(substr(p_str, 1, 7)) < '9999999' then
      -- si format de chaine en entree est valide cad 7num+cle
      -- alors on calcule la cle
      somme := 0;
      
      FOR i in 1..7 
      LOOP
        somme := somme + to_number(substr(p_str, 8 - i, 1)) * i;
      END LOOP;
      
      somme := Mod(somme, 23) + 1;
      cleMF := substr(vecteur, somme, 1);
      
      if substr(p_str, 8, 1) <> cleMF then
        return 1;
      else
        return 0;
      end if;
    else
      return 1;
    end if;
  exception
    when others then
      return 1;
  END;

  const validateRNE = (rne: string): { valid: boolean; message: string } => {
    if (!rne) {
      return { valid: true, message: '' }; // Vide = pas de validation
    }

    // Le RNE doit avoir exactement 8 caractères
    if (rne.length !== 8) {
      return { 
        valid: false, 
        message: 'Le numéro RNE doit contenir exactement 8 caractères' 
      };
    }

    // Les 7 premiers caractères doivent être des chiffres
    const sevenDigits = rne.substring(0, 7);
    if (!/^\d{7}$/.test(sevenDigits)) {
      return { 
        valid: false, 
        message: 'Les 7 premiers caractères doivent être des chiffres' 
      };
    }

    // Vérifier que les 7 chiffres sont < 9999999
    const numericValue = parseInt(sevenDigits, 10);
    if (numericValue >= 9999999) {
      return { 
        valid: false, 
        message: 'Le numéro RNE est invalide (valeur trop grande)' 
      };
    }

    // Calcul de la clé de contrôle (algorithme BCT)
    const vecteur = 'ABCDEFGHJKLMNPQRSTVWXYZ'; // 23 lettres (sans I, O, U)
    let somme = 0;

    // Calcul : somme des (chiffre * position) pour positions 1 à 7
    for (let i = 1; i <= 7; i++) {
      const digit = parseInt(rne.charAt(8 - i), 10);
      somme += digit * i;
    }

    // La clé est déterminée par (somme modulo 23)
    const indexCle = (somme % 23);
    const cleCalculee = vecteur.charAt(indexCle);
    const cleSaisie = rne.charAt(7).toUpperCase();

    if (cleSaisie !== cleCalculee) {
      return { 
        valid: false, 
        message: `Clé de contrôle incorrecte. Attendu: ${cleCalculee}, trouvé: ${cleSaisie}` 
      };
    }

    return { valid: true, message: '' };
  };

│ 🔍 DIFFÉRENCE CLÉS : INDEX CALCULATION                                       │

  ⚠️ DIFFÉRENCE APPARENTE
  ───────────────────────
  
  PL/SQL  : somme := Mod(somme, 23) + 1;
            cleMF := substr(vecteur, somme, 1);
  
  JavaScript : const indexCle = (somme % 23);
               const cleCalculee = vecteur.charAt(indexCle);
  
  🤔 POURQUOI LA DIFFÉRENCE ?
  ──────────────────────────
  
  PL/SQL utilise l'indexation 1-based :
  • substr('ABCDEFGH...', 1, 1) → 'A' (1er caractère)
  • substr('ABCDEFGH...', 13, 1) → 'M' (13ème caractère)
  
  JavaScript utilise l'indexation 0-based :
  • charAt(0) → 'A' (1er caractère)
  • charAt(12) → 'M' (13ème caractère)
  
  🎯 ÉQUIVALENCE MATHÉMATIQUE
  ───────────────────────────
  
  Pour accéder au même caractère :
  • PL/SQL : index = Mod(somme, 23) + 1  →  résultat entre 1 et 23
              puis substr(vecteur, index, 1)
  
  • JS : index = somme % 23              →  résultat entre 0 et 22
         puis charAt(index)
  
  Les deux donnent le MÊME caractère !
  
  ✅ VÉRIFICATION AVEC EXEMPLE 1695881M
  ─────────────────────────────────────
  
  Somme = 149
  
  PL/SQL :
  • Mod(149, 23) + 1 = 12 + 1 = 13
  • substr('ABCDEFGHJKLMNPQRSTVWXYZ', 13, 1) = 'M'
    (le 13ème caractère en 1-based)
  
  JavaScript :
  • 149 % 23 = 12
  • charAt(12) = 'M'
    (le caractère à l'index 12 en 0-based, donc le 13ème caractère)
  
  ✅ RÉSULTAT IDENTIQUE : 'M'

  Fonction : handleNoPieceClientBlur()
  
  const handleNoPieceClientBlur = () => {
    const noPiece = formData.noPieceClient;
    
    if (!noPiece) {
      setRneError('');
      setClientInfo(null);
      setClientNotFound(false);
      return;
    }

    // 1. Validation du format RNE (algorithme BCT)
    const validation = validateRNE(noPiece);
    
    if (!validation.valid) {
      setRneError(validation.message);
      setClientInfo(null);
      setClientNotFound(false);
      return;
    }

    // 2. Si format valide → rechercher le client automatiquement
    setRneError('');
    searchClient(formData.typePieceClient, noPiece);
  };
  
  3️⃣ VALIDATION MATRICULE FISCAL
  ───────────────────────────────
  
  ✅ Si VALIDE → Passer à l'étape 4
  ❌ Si INVALIDE → Afficher message d'erreur (ARRÊT)
  
  4️⃣ RECHERCHE CLIENT VIA API
  ────────────────────────────
  
  const searchClient = async (typePiece, noPiece) => {
    setSearchingClient(true);
    setClientNotFound(false);
    setClientInfo(null);
    
    try {
      const response = await fetch(
        `/api/ref/clients/search/byPiece?typePiece=${typePiece}&noPiece=${noPiece}`
      );
      
      if (response.ok) {
        const data: ClientInfo = await response.json();
        setClientInfo(data);
        setClientNotFound(false);
        // ✅ Affichage nom client + liste des comptes
      } else if (response.status === 404) {
        setClientInfo(null);
        setClientNotFound(true);
        toast.warning('Client non trouvé');
      }
    } finally {
      setSearchingClient(false);
    }
  };
