  implémente une VALIDATION AUTOMATIQUE avec RECHERCHE CLIENT intégrée.

  ✅ Option de recherche manuelle via bouton 🔍

┌──────────────────────────────────────────────────────────────────────────────┐
│ 📋 ALGORITHME DE VALIDATION RNE                                              │
└──────────────────────────────────────────────────────────────────────────────┘

  🔐 ALGORITHME BCT (Banque Centrale de Tunisie)
  ──────────────────────────────────────────────
  
  Le numéro RNE tunisien suit un format strict avec clé de contrôle :
  
  📏 FORMAT : 7 chiffres + 1 lettre de contrôle
  
  Exemple : 1695881M
            ─┬────┘ └─ Clé de contrôle (lettre)
             └──────── 7 chiffres

  🧮 RÈGLES DE VALIDATION
  ───────────────────────
  
  1️⃣ Longueur exacte : 8 caractères
  
  2️⃣ Les 7 premiers caractères : chiffres uniquement (0-9)
  
  3️⃣ Valeur numérique : les 7 chiffres < 9999999
  
  4️⃣ Le 8ème caractère : lettre de contrôle calculée
  
  🔢 CALCUL DE LA CLÉ
  ────────────────────
  
  Vecteur de clés : "ABCDEFGHJKLMNPQRSTVWXYZ" (23 lettres, sans I, O, U)
  
  Algorithme :
  
  1. Pour chaque chiffre (position 1 à 7), calculer :
     somme += chiffre × position_inverse
     
     Position inverse : (8 - position_dans_la_chaine)
     
     Exemple pour 1695881 :
     Position :     1   2   3   4   5   6   7
     Chiffre  :     1   6   9   5   8   8   1
     Poids    :     7   6   5   4   3   2   1
     Produit  :     7  36  45  20  24  16   1
     
     Somme totale = 7 + 36 + 45 + 20 + 24 + 16 + 1 = 149
  
  2. Index de la clé = (somme % 23)
     
     149 % 23 = 12
  
  3. Clé = vecteur[index]
     
     vecteur[12] = 'M' (A=0, B=1, C=2, ..., M=12)
  
  4. RNE final = "1695881M" ✓

  📝 EXEMPLES DE VALIDATION
  ──────────────────────────
  
  ✅ VALIDE : 1695881M
     • 7 chiffres : 1695881 ✓
     • < 9999999 ✓
     • Clé calculée : M ✓
     • Clé saisie : M ✓
  
  ❌ INVALIDE : 1695881A
     • Clé calculée : M
     • Clé saisie : A
     • Message : "Clé de contrôle incorrecte. Attendu: M, trouvé: A"
  
  ❌ INVALIDE : 123456
     • Longueur : 6 caractères
     • Message : "Le numéro RNE doit contenir exactement 8 caractères"
  
  ❌ INVALIDE : 169588AB
     • Position 7 : lettre au lieu de chiffre
     • Message : "Les 7 premiers caractères doivent être des chiffres"

  ──────────────────
  
  • Désactivé (gris) si :
    - Champ vide
    - Recherche en cours
  
  • Actif (bleu) si :
    - Champ rempli
    - Pas de recherche en cours
  
  • Icône :
    - 🔍 (loupe) : état normal
    - ⏳ (spinner) : recherche en cours

  📌 FONCTION : validateRNE(rne: string)
  ──────────────────────────────────────
  
  Retourne : { valid: boolean; message: string }
  
  const validateRNE = (rne: string): { valid: boolean; message: string } => {
    if (!rne) return { valid: true, message: '' };
    
    // Validation longueur
    if (rne.length !== 8) {
      return { 
        valid: false, 
        message: 'Le numéro RNE doit contenir exactement 8 caractères' 
      };
    }
    
    // Validation format (7 chiffres)
    const sevenDigits = rne.substring(0, 7);
    if (!/^\d{7}$/.test(sevenDigits)) {
      return { 
        valid: false, 
        message: 'Les 7 premiers caractères doivent être des chiffres' 
      };
    }
    
    // Validation valeur numérique
    const numericValue = parseInt(sevenDigits, 10);
    if (numericValue >= 9999999) {
      return { 
        valid: false, 
        message: 'Le numéro RNE est invalide (valeur trop grande)' 
      };
    }
    
    // Calcul et validation clé
    const vecteur = 'ABCDEFGHJKLMNPQRSTVWXYZ';
    let somme = 0;
    
    for (let i = 1; i <= 7; i++) {
      const digit = parseInt(rne.charAt(8 - i), 10);
      somme += digit * i;
    }
    
    const indexCle = (somme % 23);
    const cleCalculee = vecteur.charAt(indexCle);
    const cleSaisie = rne.charAt(7).toUpperCase();
    
    if (cleSaisie !== cleCalculee) {
      return { 
        valid: false, 
        message: `Clé de contrôle incorrecte. Attendu: ${cleCalculee}, trouvé: ${cleSaisie}` 
      };
    }
    
    return { valid: true, message: '' };
  };
  
  📌 FONCTION : handleNoPieceClientBlur()
  ───────────────────────────────────────
  
  const handleNoPieceClientBlur = () => {
    const noPiece = formData.noPieceClient;
    
    if (!noPiece) {
      setRneError('');
      setClientInfo(null);
      setClientNotFound(false);
      return;
    }

    // Validation RNE
    const validation = validateRNE(noPiece);
    
    if (!validation.valid) {
      setRneError(validation.message);
      setClientInfo(null);
      setClientNotFound(false);
      return;
    }

    // Recherche automatique
    setRneError('');
    searchClient(formData.typePieceClient, noPiece);
  };
  
  📌 FONCTION : searchClient(typePiece, noPiece)
  ──────────────────────────────────────────────
  
  const searchClient = async (typePiece, noPiece) => {
    setSearchingClient(true);
    setClientNotFound(false);
    setClientInfo(null);
    
    try {
      const response = await fetch(
        `/api/ref/clients/search/byPiece?typePiece=${typePiece}&noPiece=${noPiece}`
      );
      
      if (response.ok) {
        const data = await response.json();
        setClientInfo(data);
        setClientNotFound(false);
      } else if (response.status === 404) {
        setClientInfo(null);
        setClientNotFound(true);
        toast.warning('Client non trouvé');
      }
    } catch (error) {
      // Mode démonstration avec données mock
      setClientInfo(mockData);
    } finally {
      setSearchingClient(false);
    }
  };
