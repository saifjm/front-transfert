  POURQUOI PAS DE "+1" EN JAVASCRIPT ?
  ─────────────────────────────────────
  
  PL/SQL (indexation 1-based) :
  • Mod(149, 23) = 11
  • index = 11 + 1 = 12
  • substr(vecteur, 12, 1) = 'M' (12ème caractère, 1-based)
  
  JavaScript (indexation 0-based) :
  • 149 % 23 = 11
  • index = 11 (pas de +1 !)
  • charAt(11) = 'M' (caractère à l'index 11, 0-based)
  
  🎯 RÉSULTAT IDENTIQUE
  ─────────────────────
  
  Les deux approches donnent 'M' car :
  • Position 12 en 1-based = Index 11 en 0-based
  • Le "+1" du PL/SQL compense la différence d'indexation
  
  C'est pourquoi l'implémentation JavaScript n'a PAS BESOIN du "+1" !

  const validateRNE = (rne: string): { valid: boolean; message: string } => {
    if (!rne) {
      return { valid: true, message: '' }; // Vide = pas de validation
    }

    // Le RNE doit avoir exactement 8 caractères
    if (rne.length !== 8) {
      return { 
        valid: false, 
        message: 'Le numéro RNE doit contenir exactement 8 caractères' 
      };
    }

    // Les 7 premiers caractères doivent être des chiffres
    const sevenDigits = rne.substring(0, 7);
    if (!/^\d{7}$/.test(sevenDigits)) {
      return { 
        valid: false, 
        message: 'Les 7 premiers caractères doivent être des chiffres' 
      };
    }

    // Vérifier que les 7 chiffres sont < 9999999
    const numericValue = parseInt(sevenDigits, 10);
    if (numericValue >= 9999999) {
      return { 
        valid: false, 
        message: 'Le numéro RNE est invalide (valeur trop grande)' 
      };
    }

    // Calcul de la clé de contrôle (algorithme BCT)
    const vecteur = 'ABCDEFGHJKLMNPQRSTVWXYZ'; // 23 lettres (sans I, O, U)
    let somme = 0;

    // Calcul : somme des (chiffre * position) pour positions 1 à 7
    for (let i = 1; i <= 7; i++) {
      const digit = parseInt(rne.charAt(8 - i), 10);
      somme += digit * i;
    }

    // La clé est déterminée par (somme modulo 23)
    // En PL/SQL : Mod(somme, 23) + 1 puis substr(vecteur, index, 1) [1-based]
    // En JavaScript : somme % 23 puis charAt(index) [0-based]
    // Les deux donnent le même résultat car charAt(11) = substr(..., 12, 1)

    const cleCalculee = vecteur.charAt(indexCle);
    const cleSaisie = rne.charAt(7).toUpperCase();

      return { 
        valid: false, 
        message: `Clé de contrôle incorrecte. Attendu: ${cleCalculee}, trouvé: ${cleSaisie}` 
      };
    }

    return { valid: true, message: '' };
  };
